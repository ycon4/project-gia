import React, { useState, useMemo, useRef, useEffect } from 'react';
import { 
  Plus, Calendar, Upload, BarChart3, Trash2, MoreVertical,
  Download, CheckCircle2, X, Search, Edit3, PlusSquare,
  Users, UserCheck, ShieldAlert, CalendarCheck, Fingerprint, Link2
} from 'lucide-react';
import { QRCodeSVG } from 'qrcode.react';
import * as XLSX from 'xlsx';
//import { 
//  BarChart, Bar, XAxis, YAxis, CartesianGrid, 
//  Tooltip, Legend, ResponsiveContainer, PieChart, Pie, Cell,
 // AreaChart, Area, LineChart, Line
//} from 'recharts';
import GeneralDashboard from '../components/events/GeneralPage';

const COLORS = ['#6366f1', '#ec4899', '#8b5cf6', '#10b981', '#f59e0b', '#06b6d4'];

const SectionTitle = ({ children, desc }) => (
  <div className="mb-4">
    <h3 className="text-[10px] font-black text-slate-400 uppercase tracking-widest leading-none">{children}</h3>
    {desc && <p className="text-[9px] text-slate-300 font-bold mt-1 uppercase">{desc}</p>}
  </div>
);

export default function EventsPage({ 
  events = [], 
  attendanceData = [], 
  onCreateEvent = () => {},
  onDeleteEvent = () => {},
  onUpdateEvent = () => {},
  onBulkUpload = () => {}
}) {
  const [activeEvent, setActiveEvent] = useState(null);
  const [selectedSession, setSelectedSession] = useState('');
  const [showCreateModal, setShowCreateModal] = useState(false);
  const [showActionMenu, setShowActionMenu] = useState(false);
  const [searchTerm, setSearchTerm] = useState('');
  const [isEditing, setIsEditing] = useState(false);
  const fileInputRef = useRef(null);
  
  const [newEvent, setNewEvent] = useState({ 
    title: '', description: '', status: 'Active',
    sessions: ['Pre-Registration', 'Day 1 Attendance'],
    formConfig: { sex: true, office_college: true, pwd_status: true, sector: true }
  });

  // --- 1. THE FIX: SYNC SESSION STATE WHEN EVENT CHANGES ---
  useEffect(() => {
    if (activeEvent) {
      // If the currently selected session isn't in the new active event's list, reset it
      if (!activeEvent.sessions?.includes(selectedSession)) {
        setSelectedSession(activeEvent.sessions?.[0] || '');
      }
    }
  }, [activeEvent]);

  // --- 2. ANALYTICS ENGINE (Robust Filters) ---
  const stats = useMemo(() => {
    if (!activeEvent) return null;

    const eventAttendance = attendanceData.filter(
      record => String(record.eventId) === String(activeEvent.id)
    );

    const sdd = {
      Male: eventAttendance.filter(r => r.sex === 'Male').length,
      Female: eventAttendance.filter(r => r.sex === 'Female').length,
    };

    const sessionData = (activeEvent.sessions || []).map(sessionName => ({
      name: sessionName.replace(' Attendance', ''),
      Male: eventAttendance.filter(r => r.session === sessionName && r.sex === 'Male').length,
      Female: eventAttendance.filter(r => r.session === sessionName && r.sex === 'Female').length,
      total: eventAttendance.filter(r => r.session === sessionName).length
    }));

    const sectorMap = eventAttendance.reduce((acc, curr) => {
      const s = curr.sector || 'Others';
      acc[s] = (acc[s] || 0) + 1;
      return acc;
    }, {});

    const sectorData = Object.keys(sectorMap).map(key => ({ name: key, value: sectorMap[key] }));
    const pwdCount = eventAttendance.filter(r => r.pwd_status === 'Yes' || r.pwd_status === 'PWD').length;
    const inclusionData = [{ name: 'PWD', value: pwdCount }, { name: 'Non-PWD', value: eventAttendance.length - pwdCount }];
    
    const officeMap = eventAttendance.reduce((acc, curr) => {
      const o = curr.office_college || 'N/A';
      acc[o] = (acc[o] || 0) + 1;
      return acc;
    }, {});
    const officeData = Object.keys(officeMap).map(key => ({ name: key, value: officeMap[key] })).sort((a,b) => b.value - a.value).slice(0, 5);

    return { eventAttendance, sdd, sessionData, sectorData, inclusionData, officeData };
  }, [activeEvent, attendanceData]);

  const filteredAttendance = useMemo(() => {
      if (!stats) return [];
      
      return stats.eventAttendance.filter(item => {
        // 1. Search logic
        const matchesSearch = !searchTerm || 
          item.fullName?.toLowerCase().includes(searchTerm.toLowerCase()) ||
          item.office_college?.toLowerCase().includes(searchTerm.toLowerCase());    

        // 2. Fuzzy Session logic
        // We remove the word "attendance" and "registration" to compare just the core name
        const clean = (str) => String(str || '')
          .toLowerCase()
          .replace('attendance', '')
          .replace('registration', '')
          .replace('pre-', '')
          .trim();

        const itemSess = clean(item.session);
        const selectedSess = clean(selectedSession);
        
        // This allows "Day 1" to match "DAY 1 ATTENDANCE"
        const matchesSession = itemSess === selectedSess || 
                              itemSess.includes(selectedSess) || 
                              selectedSess.includes(itemSess);
        
        return matchesSearch && matchesSession;
      });
    }, [stats, searchTerm, selectedSession]);

  // --- 3. ACTIONS ---
  const handleEditClick = () => {
    setNewEvent({ ...activeEvent });
    setIsEditing(true);
    setShowCreateModal(true);
    setShowActionMenu(false);
  };

  const handleAddDay = () => {
    const currentSessions = activeEvent.sessions || [];
    const nextDayNum = currentSessions.filter(s => s.toLowerCase().includes('day')).length + 1;
    const newSession = `Day ${nextDayNum} Attendance`;
    
    const updatedEvent = { 
      ...activeEvent, 
      sessions: [...currentSessions, newSession] 
    };

    // Update parent/database first
    onUpdateEvent(activeEvent.id, updatedEvent);
    // Update local state immediately with the same object
    setActiveEvent(updatedEvent);
    setSelectedSession(newSession);
    setShowActionMenu(false);
  };

  const handleToggleStatus = () => {
    const updatedEvent = { ...activeEvent, status: activeEvent.status === 'Done' ? 'Active' : 'Done' };
    onUpdateEvent(activeEvent.id, updatedEvent);
    setActiveEvent(updatedEvent);
    setShowActionMenu(false);
  };

  const handleExcelUpload = (e) => {
    const file = e.target.files[0];
    if (!file || !selectedSession || !activeEvent) return;
    const reader = new FileReader();
    reader.onload = (evt) => {
      try {
        const wb = XLSX.read(evt.target.result, { type: 'binary' });
        const ws = wb.Sheets[wb.SheetNames[0]];
        const data = XLSX.utils.sheet_to_json(ws);
        const formattedData = data.filter(row => row.Name || row['Full Name']).map(row => ({
          eventId: activeEvent.id,
          fullName: String(row.Name || row['Full Name']).trim(),
          session: selectedSession,
          sex: row.Sex,
          sector: row.Sector,
          office_college: row['Office / College'],
          pwd_status: row.PWD || 'No',
          timestamp: new Date().toISOString()
        }));
        onBulkUpload(formattedData);
        e.target.value = null;
      } catch (err) { alert("Import Error."); }
    };
    reader.readAsBinaryString(file);
  };

  const registrationUrl = activeEvent ? `${window.location.origin}/register?event=${activeEvent.id}&session=${encodeURIComponent(selectedSession)}` : '';
 
  console.log("Current Session State:", selectedSession);
  console.log("First Data Record Session:", stats?.eventAttendance[0]?.session);

  return (
    <div className="space-y-6 pb-10">
      {/* MODAL & HEADER CODE (Keep same as your provided draft) */}
      {showCreateModal && (
        <div className="fixed inset-0 bg-slate-900/60 backdrop-blur-md z-[70] flex items-center justify-center p-4">
          <div className="bg-white rounded-[40px] w-full max-w-lg shadow-2xl max-h-[90vh] overflow-y-auto">
            <div className="sticky top-0 bg-white z-20 px-10 pt-10 pb-4 flex justify-between items-center">
              <h2 className="text-xl font-black text-slate-900 uppercase">{isEditing ? 'Update GAD Activity' : 'Initialize GAD Activity'}</h2>
              <button onClick={() => { setShowCreateModal(false); setIsEditing(false); }} className="p-2 hover:bg-slate-100 rounded-full text-slate-400"><X size={20}/></button>
            </div>
            <div className="px-10 pb-10 pt-2">
              <form onSubmit={(e) => {
                  e.preventDefault();
                  isEditing ? onUpdateEvent(activeEvent.id, newEvent) : onCreateEvent(newEvent);
                  alert(isEditing ? "Activity updated successfully!" : "Activity has been successfully created!")
                  setShowCreateModal(false);
                  setIsEditing(false);
                }} className="space-y-6">
                  
                  {/* Basic Information Section */}
                  <div className="space-y-4">
                    <h3 className="text-xs font-semibold text-slate-500 uppercase tracking-wider border-b pb-2">Basic Information</h3>
                    
                    <div>
                      <label className="block text-sm font-medium text-slate-700 mb-1">Event Title</label>
                      <input 
                        required 
                        className="w-full border border-slate-300 rounded-md px-3 py-2 text-sm focus:ring-1 focus:ring-blue-500 outline-none" 
                        placeholder="e.g. Annual GAD Symposium"
                        value={newEvent.title} 
                        onChange={e => setNewEvent({...newEvent, title: e.target.value})} 
                      />
                    </div>

                    <div className="grid grid-cols-2 gap-4">
                      <div>
                        <label className="block text-sm font-medium text-slate-700 mb-1">Start Date</label>
                        <input 
                          type="date"
                          required 
                          className="w-full border border-slate-300 rounded-md px-3 py-2 text-sm focus:ring-1 focus:ring-blue-500 outline-none" 
                          value={newEvent.startDate || ''} 
                          onChange={e => setNewEvent({...newEvent, startDate: e.target.value})} 
                        />
                      </div>
                      <div>
                        <label className="block text-sm font-medium text-slate-700 mb-1">End Date</label>
                        <input 
                          type="date"
                          required 
                          className="w-full border border-slate-300 rounded-md px-3 py-2 text-sm focus:ring-1 focus:ring-blue-500 outline-none" 
                          value={newEvent.endDate || ''} 
                          onChange={e => setNewEvent({...newEvent, endDate: e.target.value})} 
                        />
                      </div>
                    </div>

                    <div>
                      <label className="block text-sm font-medium text-slate-700 mb-1">Description</label>
                      <textarea 
                        className="w-full border border-slate-300 rounded-md px-3 py-2 text-sm focus:ring-1 focus:ring-blue-500 outline-none min-h-[80px]" 
                        placeholder="Brief overview of the activity..."
                        value={newEvent.description} 
                        onChange={e => setNewEvent({...newEvent, description: e.target.value})} 
                      />
                    </div>
                  </div>

                  {/* NEW: Pre-registration Toggle Section */}
                  <div className="space-y-4">
                    <h3 className="text-xs font-semibold text-slate-500 uppercase tracking-wider border-b pb-2">Registration Workflow</h3>
                    <div className="flex items-center justify-between bg-blue-50 p-4 rounded-lg border border-blue-100">
                      <div>
                        <p className="text-sm font-bold text-blue-900">Include Pre-registration?</p>
                        <p className="text-[10px] text-blue-700">If enabled, a unique QR code for online pre-registration will be generated.</p>
                      </div>
                      <div className="flex bg-white rounded-md p-1 border border-blue-200">
                        <button 
                          type="button"
                          onClick={() => setNewEvent({...newEvent, hasPreReg: true})}
                          className={`px-4 py-1.5 text-xs font-bold rounded ${newEvent.hasPreReg ? 'bg-blue-600 text-white' : 'text-slate-400'}`}
                        >
                          YES
                        </button>
                        <button 
                          type="button"
                          onClick={() => setNewEvent({...newEvent, hasPreReg: false})}
                          className={`px-4 py-1.5 text-xs font-bold rounded ${!newEvent.hasPreReg ? 'bg-slate-600 text-white' : 'text-slate-400'}`}
                        >
                          NO
                        </button>
                      </div>
                    </div>
                  </div>

                  {/* Form Attributes Configuration */}
                  <div className="space-y-4">
                    <div className="flex justify-between items-end border-b pb-2">
                      <h3 className="text-xs font-semibold text-slate-500 uppercase tracking-wider">Attendance Form Attributes</h3>
                      <span className="text-[10px] text-slate-400 italic">Select fields to include in the registration form</span>
                    </div>

                    <div className="grid grid-cols-2 sm:grid-cols-3 gap-3 bg-slate-50 p-4 rounded-lg border border-slate-200">
                      {[
                        { id: 'fullName', label: 'Full Name' },
                        { id: 'sex', label: 'Sex/Gender' },
                        { id: 'age', label: 'Age' },
                        { id: 'home_address', label: 'Home Address' },
                        { id: 'email', label: 'Email Address' },
                        { id: 'phone', label: 'Phone Number' },
                        { id: 'office_college', label: 'College/Office' },
                        { id: 'department', label: 'Department' },
                        { id: 'designation', label: 'Designation' },
                        { id: 'sector', label: 'Sector' },
                        { id: 'pwd_status', label: 'PWD Status' },
                        { id: 'ethnic_group', label: 'Ethnic Group' },
                        { id: 'employment_status', label: 'Employment Status' },
                        { id: 'year_level', label: 'Year Level (Students)' },
                        { id: 'emergency_contact', label: 'Emergency Contact' }
                      ].map((attr) => (
                        <label key={attr.id} className="flex items-center gap-2 cursor-pointer group">
                          <input 
                            type="checkbox" 
                            className="w-4 h-4 rounded border-slate-300 text-blue-600 focus:ring-blue-500"
                            checked={newEvent.formConfig?.[attr.id] || false}
                            onChange={(e) => setNewEvent({
                              ...newEvent, 
                              formConfig: { ...newEvent.formConfig, [attr.id]: e.target.checked }
                            })}
                          />
                          <span className="text-xs text-slate-600 group-hover:text-slate-900 transition-colors">
                            {attr.label}
                          </span>
                        </label>
                      ))}
                    </div>
                  </div>

                  <div className="pt-4">
                    <button 
                      type="submit" 
                      className="w-full py-3 bg-blue-700 text-white rounded-md font-semibold text-sm hover:bg-blue-800 transition-all shadow-sm flex items-center justify-center gap-2"
                    >
                      {isEditing ? <Edit3 size={16}/> : <PlusSquare size={16}/>}
                      {isEditing ? 'Update Activity' : 'Initialize Activity'}
                    </button>
                  </div>
              </form>
            </div>
          </div>
        </div>
      )}

      <div className="flex flex-col md:flex-row justify-between items-center bg-white p-6 rounded-[32px] border border-slate-100 shadow-sm gap-4">
        <div className="flex items-center gap-4">
          <div className="w-12 h-12 bg-indigo-600 rounded-2xl flex items-center justify-center text-white shadow-lg"><Calendar size={24} /></div>
          <div>
            <h1 className="text-2xl font-black text-slate-900 uppercase tracking-tight leading-none">GAD Event Manager</h1>
            <p className="text-[9px] font-bold text-slate-400 uppercase tracking-widest mt-1">Real-time SDD Tracking</p>
          </div>
        </div>
        <button onClick={() => { setNewEvent({title:'', description:'', status:'Active', sessions:['Pre-Registration', 'Day 1 Attendance'], formConfig: newEvent.formConfig}); setIsEditing(false); setShowCreateModal(true); }} className="px-6 py-3 bg-slate-900 text-white rounded-2xl font-black text-[10px] uppercase tracking-widest flex items-center gap-2 hover:bg-indigo-600 transition-all">
          <Plus size={16}/> Create Event
        </button>
      </div>

      <div className="grid grid-cols-1 lg:grid-cols-12 gap-6">
        {/* SIDEBAR */}
        <div className="lg:col-span-3 space-y-3">
          <button onClick={() => setActiveEvent(null)} className={`w-full p-5 rounded-[28px] border transition-all flex items-center gap-3 ${!activeEvent ? 'bg-slate-900 border-slate-900 text-white shadow-xl shadow-slate-200' : 'bg-white border-slate-100 hover:bg-slate-50'}`}>
            <BarChart3 size={18}/><span className="text-[11px] font-black uppercase tracking-widest">Global Dashboard</span>
          </button>
          <div className="h-px bg-slate-100 my-4" />
          <div className="space-y-2 max-h-[60vh] overflow-y-auto pr-2 custom-scrollbar">
            {events.map(event => (
              <div key={event.id} onClick={() => 
              { setActiveEvent(event); 
                setSelectedSession(event.sessions?.[0] || ''); 
              }} className={`p-4 rounded-2xl border cursor-pointer transition-all ${activeEvent?.id === event.id ? 'border-indigo-500 bg-indigo-50/30' : 'bg-white border-slate-100 hover:bg-slate-50'}`}>
                <h4 className="text-[11px] font-black text-slate-900 uppercase truncate mb-1">{event.title}</h4>
                <div className="flex justify-between items-center text-[8px] font-black text-slate-400 uppercase">
                  <span>{event.status}</span>
                  {event.status === 'Done' && <CheckCircle2 size={12} className="text-emerald-500"/>}
                </div>
              </div>
            ))}
          </div>
        </div>

        {/* MAIN CONTENT */}
        <div className="lg:col-span-9">
          {!activeEvent ? (
            <GeneralDashboard events={events} attendanceData={attendanceData} />
          ) : (
            <div className="space-y-6 animate-in slide-in-from-bottom-4 duration-500">
              {/* TOP METRICS */}
              <div className="grid grid-cols-2 md:grid-cols-4 gap-4">
                {[
                  { label: 'Total Attendance', val: stats.eventAttendance.length, icon: Users, col: 'text-slate-900' },
                  { label: 'Female Count', val: stats.sdd.Female, icon: UserCheck, col: 'text-pink-600' },
                  { label: 'Male Count', val: stats.sdd.Male, icon: UserCheck, col: 'text-indigo-600' },
                  { label: 'PWD Count', val: stats.inclusionData[0].value, icon: ShieldAlert, col: 'text-emerald-600' }
                ].map((m, idx) => (
                  <div key={idx} className="bg-white p-5 rounded-[32px] border border-slate-100 shadow-sm">
                    <m.icon size={16} className={`${m.col} mb-2`}/>
                    <p className="text-[9px] font-black text-slate-400 uppercase">{m.label}</p>
                    <p className={`text-2xl font-black ${m.col}`}>{m.val}</p>
                  </div>
                ))}
              </div>        

              {/* QR & ACTION BAR */}
                <div className="bg-indigo-950 p-8 rounded-3xl text-white flex flex-col md:flex-row items-center gap-10 shadow-xl border border-indigo-800">
                  <div className="flex-1 space-y-4">
                    <div className="space-y-1">
                      <h3 className="text-2xl font-bold leading-tight flex items-center gap-2">
                        {selectedSession === 'Pre-Registration' ? <CalendarCheck className="text-indigo-400" /> : <Fingerprint className="text-indigo-400" />}
                        {selectedSession === 'Pre-Registration' ? 'Pre-Registration' : 'Live Session'} QR
                      </h3>
                      <p className="text-indigo-300 text-xs font-medium">
                        {selectedSession === 'Pre-Registration' 
                          ? 'Share this for advanced sign-ups.' 
                          : 'Use this for on-site attendance marking.'}
                      </p>
                    </div>

                    <div className="flex flex-col gap-2">
                      <label className="text-[10px] font-bold text-indigo-400 uppercase tracking-widest ml-1">Select Active Session</label>
                      <select 
                        className="bg-indigo-900 border border-indigo-700 rounded-xl p-3.5 text-sm font-semibold outline-none focus:ring-2 focus:ring-indigo-500 transition-all appearance-none cursor-pointer" 
                        value={selectedSession} 
                        onChange={e => setSelectedSession(e.target.value)}
                      >
                        {/* Only show Pre-Registration if enabled in settings */}
                        {activeEvent.hasPreReg && (
                          <option value="Pre-Registration">📅 Pre-Registration</option>
                        )}
                        
                        {/* Dynamic Sessions (Day 1, Day 2, etc.) */}
                        {activeEvent.sessions?.map(s => (
                          <option key={s} value={s}>📍 {s}</option>
                        ))}
                      </select>
                    </div>
                  </div>

                  {/* QR Code Container */}
                  <div className="bg-white p-5 rounded-2xl shadow-2xl transition-transform hover:scale-105 duration-300">
                    <QRCodeSVG 
                      value={registrationUrl} 
                      size={150} 
                      level="H" 
                      includeMargin={false} 
                      fgColor="#1e1b4b" // Matching indigo-950
                    />
                    <div className="mt-3 text-center">
                        <span className="text-[9px] font-black text-slate-400 uppercase tracking-tighter">Scan to Register</span>
                    </div>
                  </div>

                  {/* Actions */}
                  <div className="flex flex-col gap-3 w-full md:w-auto">
                    <button 
                      onClick={() => setShowActionMenu(!showActionMenu)} 
                      className="flex items-center justify-center gap-3 bg-indigo-500/10 border border-indigo-500/30 px-6 py-4 rounded-xl font-bold text-xs hover:bg-indigo-500/20 transition-all"
                    >
                      <MoreVertical size={18}/> Form Settings
                    </button>
                    
                    <button 
                      onClick={() => { 
                        navigator.clipboard.writeText(registrationUrl); 
                        alert(`${selectedSession} Link Copied!`); 
                      }} 
                      className="flex items-center justify-center gap-3 bg-white text-indigo-950 px-6 py-4 rounded-xl font-bold text-xs hover:bg-indigo-50 transition-all shadow-lg"
                    >
                      <Link2 size={18}/> Copy {selectedSession === 'Pre-Registration' ? 'Pre-Reg' : 'Session'} Link
                    </button>
                  </div>
                </div>
              {/* SETTINGS MENU POPOVER */}
              {showActionMenu && (
                <div className="fixed inset-0 z-50 flex items-center justify-center p-4 bg-slate-900/20 backdrop-blur-sm" onClick={() => setShowActionMenu(false)}>
                  <div className="bg-white rounded-[32px] p-4 w-64 shadow-2xl border border-slate-100 space-y-2" onClick={e => e.stopPropagation()}>
                    <button onClick={handleEditClick} className="w-full text-left px-4 py-3 hover:bg-slate-50 rounded-xl flex items-center gap-3 text-slate-600 text-[10px] font-black uppercase"><Edit3 size={16}/> Update Activity</button>
                    <button onClick={handleAddDay} className="w-full text-left px-4 py-3 hover:bg-slate-50 rounded-xl flex items-center gap-3 text-indigo-600 text-[10px] font-black uppercase"><PlusSquare size={16}/> Add Day</button>
                    <button onClick={handleToggleStatus} className="w-full text-left px-4 py-3 hover:bg-slate-50 rounded-xl flex items-center gap-3 text-emerald-600 text-[10px] font-black uppercase"><CheckCircle2 size={16}/> Mark as {activeEvent.status === 'Done' ? 'Active' : 'Done'}</button>
                    <div className="h-px bg-slate-100 my-1" />
                    <button onClick={() => { onDeleteEvent(activeEvent.id); setActiveEvent(null); }} className="w-full text-left px-4 py-3 hover:bg-red-50 rounded-xl flex items-center gap-3 text-red-500 text-[10px] font-black uppercase"><Trash2 size={16}/> Delete</button>
                  </div>
                </div>
              )}

              {/* DATA TABLE */}
              <div className="bg-white rounded-[40px] border border-slate-100 overflow-hidden shadow-sm">
                <div className="p-6 border-b border-slate-50 flex flex-col md:flex-row justify-between items-center gap-4 bg-slate-50/30">
                  <div className="bg-indigo-100 text-indigo-600 px-3 py-1 rounded-full text-[10px] font-black uppercase">
                      Viewing: {selectedSession}
                  </div>
                  <div className="relative flex-1 w-full">
                    <Search className="absolute left-4 top-3.5 text-slate-300" size={16} />
                    <input type="text" placeholder="Filter by Name, College, or Sector..." className="w-full bg-white border border-slate-200 rounded-2xl pl-12 py-3 text-xs font-bold outline-none focus:ring-2 focus:ring-indigo-500 transition-all" value={searchTerm} onChange={e => setSearchTerm(e.target.value)} />
                  </div>
                  <div className="flex gap-2 w-full md:w-auto">
                    <input type="file" accept=".xlsx" className="hidden" id="xl-upload" ref={fileInputRef} onChange={handleExcelUpload} />
                    <label htmlFor="xl-upload" className="flex-1 md:flex-none cursor-pointer px-6 py-3 bg-slate-900 text-white rounded-2xl text-[10px] font-black uppercase flex items-center justify-center gap-2 hover:bg-indigo-600 transition-all"><Upload size={16}/> Import</label>
                    <button onClick={() => {
                      const ws = XLSX.utils.json_to_sheet(stats.eventAttendance);
                      const wb = XLSX.utils.book_new();
                      XLSX.utils.book_append_sheet(wb, ws, "Attendance");
                      XLSX.writeFile(wb, `${activeEvent.title}_Report.xlsx`);
                    }} className="p-3 bg-emerald-50 text-emerald-600 rounded-2xl hover:bg-emerald-100 transition-colors"><Download size={18}/></button>
                  </div>
                </div>
                <div className="max-h-[500px] overflow-auto custom-scrollbar">
                  <table className="w-full text-left border-collapse">
                    <thead className="bg-white text-[10px] font-black text-slate-400 uppercase sticky top-0 z-10 shadow-sm">
                      <tr>
                        <th className="px-8 py-5">Participant Name</th>
                        <th className="px-8 py-5 text-center">Sector</th>
                        <th className="px-8 py-5 text-center">Sex / Gender</th>
                        <th className="px-8 py-5 text-center">Session</th>
                        <th className="px-8 py-5 text-center">PWD</th>
                      </tr>
                    </thead>
                    <tbody className="divide-y divide-slate-50">
                      {filteredAttendance.map((user, i) => (
                        <tr key={i} className="hover:bg-indigo-50/20 transition-colors group">
                          <td className="px-8 py-4">
                            <p className="text-xs font-bold text-slate-700">{user.fullName}</p>
                            <p className="text-[9px] text-slate-400 font-bold uppercase">{user.office_college || 'General'}</p>
                          </td>
                          <td className="px-8 py-4 text-center">
                            <span className="px-3 py-1 bg-slate-100 text-slate-600 rounded-full text-[9px] font-black uppercase group-hover:bg-white transition-colors">{user.sector || 'N/A'}</span>
                          </td>
                          <td className="px-8 py-4 text-center">
                            <span className={`text-[10px] font-black uppercase ${user.sex === 'Male' ? 'text-indigo-500' : 'text-pink-500'}`}>{user.sex}</span>
                          </td>
                          <td className="px-8 py-4 text-center">
                             <span className="text-[10px] font-bold text-slate-400">{user.session?.replace(' Attendance','')}</span>
                          </td>
                          <td className="px-8 py-4 text-center">
                             <span className={`text-[10px] font-black ${user.pwd_status === 'Yes' || user.pwd_status === 'PWD' ? 'text-emerald-500' : 'text-slate-200'}`}>
                               {user.pwd_status === 'Yes' || user.pwd_status === 'PWD' ? 'YES' : 'NO'}
                             </span>
                          </td>
                        </tr>
                      ))}
                    </tbody>
                  </table>
                  {filteredAttendance.length === 0 && (
                    <div className="py-20 text-center">
                      <Users size={40} className="mx-auto text-slate-100 mb-4"/>
                      <p className="text-xs font-black text-slate-300 uppercase tracking-widest">No participants found</p>
                    </div>
                  )}
                </div>
              </div>
            </div>
          )}
        </div>
      </div>
    </div>
  );
}
